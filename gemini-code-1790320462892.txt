dp = Dispatcher()

# --- Клавиатура ---
def get_main_keyboard() -> ReplyKeyboardMarkup:
    kb = [
        [KeyboardButton(text="О нас")],
        [KeyboardButton(text="Направления")],
        [KeyboardButton(text="Контакты")],
    ]
    return ReplyKeyboardMarkup(
        keyboard=kb,
        resize_keyboard=True,
        input_field_placeholder="Выберите интересующий раздел..."
    )

# --- Обработчики команд и сообщений ---

@dp.message(CommandStart())
async def cmd_start(message: types.Message):
    welcome_text = (
        f"Приветствуем, {message.from_user.first_name}!\n\n"
        "Добро пожаловать в бот **Молодежного ресурсного центра**!\n"
        "Мы — единая точка входа для поддержки, информирования и развития молодежи.\n\n"
        "Используйте меню ниже, чтобы узнать больше о нашей работе и возможностях."
    )
    await message.answer(
        welcome_text,
        reply_markup=get_main_keyboard(),
        parse_mode="Markdown"
    )
